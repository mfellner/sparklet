cmake_minimum_required(VERSION 3.16)
project(sparkdash_host LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(SANITIZE "Address/undefined sanitizers" ON)
if(SANITIZE)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
 add_link_options(-fsanitize=address,undefined)
endif()
set(CORE ../../firmware/sparkdash/components/core)
if(NOT DEFINED ENV{IDF_PATH})
 message(FATAL_ERROR "Activate ESP-IDF first")
endif()
add_library(json STATIC $ENV{IDF_PATH}/components/json/cJSON/cJSON.c)
target_include_directories(json PUBLIC $ENV{IDF_PATH}/components/json/cJSON)
target_compile_definitions(json PRIVATE CJSON_NESTING_LIMIT=16)
add_library(core STATIC ${CORE}/core.cpp)
target_include_directories(core PUBLIC ${CORE}/include)
target_link_libraries(core PUBLIC json)
add_executable(core_tests core_tests.cpp)
target_link_libraries(core_tests core)
enable_testing()
add_test(NAME core_tests COMMAND core_tests)
add_executable(parse_fixture parse_fixture.cpp)
target_link_libraries(parse_fixture core)
